// Возвращает начальную расстановку русских шашек (64 символа).
// б/Б — белые (простая/дамка), ч/Ч — чёрные, . — пусто (в т.ч. светлые клетки).
Функция НоваяДоска() Экспорт

	Доска = "";
	Для Строка = 0 По 7 Цикл
		Для Колонка = 0 По 7 Цикл
			Если Не ЭтоТёмнаяКлетка(Строка, Колонка) Тогда
				Доска = Доска + ".";
			ИначеЕсли Строка <= 2 Тогда
				Доска = Доска + "ч";
			ИначеЕсли Строка >= 5 Тогда
				Доска = Доска + "б";
			Иначе
				Доска = Доска + ".";
			КонецЕсли;
		КонецЦикла;
	КонецЦикла;
	Возврат Доска;

КонецФункции

Функция ЭтоТёмнаяКлетка(Строка, Колонка) Экспорт

	Возврат (Строка + Колонка) % 2 = 1;

	Индекс = Строка * 8 + Колонка + 1;

// Возвращает Структура(Успех, Строка, Колонка) для нотации вида a3.
Функция НотацияВКоординаты(Нотация) Экспорт

	Результат = Новый Структура("Успех, Строка, Колонка", Ложь, -1, -1);
	Текст = НРег(СокрЛП(Нотация));
	Если СтрДлина(Текст) < 2 Тогда
		Возврат Результат;
	КонецЕсли;
	Буква = Лев(Текст, 1);
	Цифра = Сред(Текст, 2, 1);
	Буквы = "abcdefgh";
	ПозицияБуквы = СтрНайти(Буквы, Буква);
	Если ПозицияБуквы = 0 Тогда
		Возврат Результат;
	КонецЕсли;
	Попытка
		Ряд = Число(Цифра);
	Исключение
		Возврат Результат;
	КонецПопытки;
	Если Ряд < 1 Или Ряд > 8 Тогда
		Возврат Результат;
	КонецЕсли;
	Результат.Успех = Истина;
	Результат.Колонка = ПозицияБуквы - 1;
	Результат.Строка = 8 - Ряд;
	Возврат Результат;

// Возвращает массив ходов.
// ТолькоВзятия — продолжение цепочки после взятия.
// ОбязательноеВзятие — если есть хотя бы одно взятие у стороны, тихие ходы запрещены.
Функция ПолучитьХоды(Доска, ХодБелых, ТолькоСКлеткиСтрока = -1, ТолькоСКлеткиКолонка = -1, ЗапретВзятий = "", ТолькоВзятия = Ложь, ОбязательноеВзятие = Ложь) Экспорт

	Цвет = ?(ХодБелых, "б", "ч");
	Взятия = Новый Массив;
	Тихие = Новый Массив;

	ЕстьВзятиеУСтороны = Ложь;
	Если ОбязательноеВзятие И Не ТолькоВзятия Тогда
		Для Строка = 0 По 7 Цикл
			Для Колонка = 0 По 7 Цикл
				Символ = СимволКлетки(Доска, Строка, Колонка);
				Если ЦветФигуры(Символ) <> Цвет Тогда
					Продолжить;
				КонецЕсли;
				Врем = Новый Массив;
				ДобавитьВзятияСКлетки(Доска, Строка, Колонка, Символ, ЗапретВзятий, Врем);
				Если Врем.Количество() > 0 Тогда
					ЕстьВзятиеУСтороны = Истина;
					Прервать;
				КонецЕсли;
			КонецЦикла;
			Если ЕстьВзятиеУСтороны Тогда
				Прервать;
			КонецЕсли;
		КонецЦикла;
	КонецЕсли;

	Для Строка = 0 По 7 Цикл
		Для Колонка = 0 По 7 Цикл
			Если ТолькоСКлеткиСтрока >= 0 Тогда
				Если Строка <> ТолькоСКлеткиСтрока Или Колонка <> ТолькоСКлеткиКолонка Тогда
					Продолжить;
				КонецЕсли;
			КонецЕсли;
			Символ = СимволКлетки(Доска, Строка, Колонка);
			Если ЦветФигуры(Символ) <> Цвет Тогда
				Продолжить;
			КонецЕсли;
			ДобавитьВзятияСКлетки(Доска, Строка, Колонка, Символ, ЗапретВзятий, Взятия);
			Если Не ТолькоВзятия И Не (ОбязательноеВзятие И ЕстьВзятиеУСтороны) Тогда
				ДобавитьТихиеХодыСКлетки(Доска, Строка, Колонка, Символ, Тихие);
			КонецЕсли;
		КонецЦикла;
	КонецЦикла;

	Если ТолькоВзятия Или (ОбязательноеВзятие И ЕстьВзятиеУСтороны) Тогда
		Возврат Взятия;
	КонецЕсли;

	Результат = Новый Массив;
	Для Каждого Ход Из Взятия Цикл
		Результат.Добавить(Ход);
	КонецЦикла;
	Для Каждого Ход Из Тихие Цикл
		Результат.Добавить(Ход);
	КонецЦикла;
	Возврат Результат;

		Р = СтрЗаменить(Р, "K", "");
	ИначеЕсли Символ = "r" И ИзС = 0 И ИзК = 0 Тогда
		Р = СтрЗаменить(Р, "q", "");
	ИначеЕсли Символ = "r" И ИзС = 0 И ИзК = 7 Тогда
		Р = СтрЗаменить(Р, "k", "");
	КонецЕсли;
	Если ВС = 7 И ВК = 0 Тогда
		Р = СтрЗаменить(Р, "Q", "");
	ИначеЕсли ВС = 7 И ВК = 7 Тогда

	ИначеЕсли ВС = 0 И ВК = 0 Тогда
		Р = СтрЗаменить(Р, "q", "");
	ИначеЕсли ВС = 0 И ВК = 7 Тогда
		Р = СтрЗаменить(Р, "k", "");

Функция ОбработатьКликДурака(Доска, ЯИгрок0, КодДействия, Параметр, МойИндекс = -1) Экспорт

	Результат = НовыйРезультатХодаДурака(Доска);
	Состояние = СостояниеДуракаИзСтроки(Доска);
	Если Состояние.status <> "игра" Тогда
		Результат.Описание = "Игра окончена";
		Результат.Статус = Состояние.status;
		Возврат Результат;
	КонецЕсли;

	Если МойИндекс < 0 Тогда
		Игрок = ?(ЯИгрок0, 0, 1);
	Иначе
		Игрок = МойИндекс;
	КонецЕсли;
	Код = Число(КодДействия);
	Парам = Число(Параметр);

	// Декоративные элементы (не кликабельны по смыслу).
	Если Код = 98 Тогда
		Возврат Результат;
	КонецЕсли;

	Если Состояние.turn <> Игрок Тогда
		Результат.Описание = "Сейчас ход соперника";
		Возврат Результат;
	КонецЕсли;

	Если Код = 90 Тогда
		Рука = РукаИгрокаДурака(Состояние, Игрок);
		Если Парам < 0 Или Парам >= Рука.Количество() Тогда
			Результат.Описание = "Нет такой карты";
			Возврат Результат;
		КонецЕсли;
		КодКарты = ВРег(Рука[Парам]);
		Если Состояние.sel = КодКарты Тогда
			// Повторный клик — снять выбор (не сыграть).
			Состояние.sel = "";
			Состояние.msg = "Выбор снят — выберите другую карту";
			Результат.Доска = СостояниеДуракаВСтроку(Состояние);
			Результат.Описание = Состояние.msg;
			Результат.Ок = Истина;
			Результат.Статус = Состояние.status;
		Иначе
			Состояние.sel = КодКарты;
			Состояние.msg = "Выбрана " + КодКарты + " — слот на столе, «Сыграть» или «Перевод»";
			Результат.Доска = СостояниеДуракаВСтроку(Состояние);
			Результат.Описание = Состояние.msg;
			Результат.Ок = Истина;
			Результат.Статус = Состояние.status;
		КонецЕсли;
		Возврат Результат;
	КонецЕсли;

	Если Код = 94 Тогда
		Если ПустаяСтрока(Состояние.sel) Тогда
			Результат.Описание = "Сначала выберите карту";
			Возврат Результат;
		КонецЕсли;
		Если СыгратьКартуДурака(Состояние, Игрок, Состояние.sel, Ложь) Тогда
			ЗаполнитьРезультатДурака(Результат, Состояние, Истина, "Ход: " + Состояние.sel);
		Иначе
			Результат.Описание = Состояние.msg;
			Результат.Доска = СостояниеДуракаВСтроку(Состояние);
			Результат.Ок = Истина;
		КонецЕсли;
		Возврат Результат;
	КонецЕсли;

	Если Код = 93 Тогда
		Если Не Состояние.perevod Тогда
			Результат.Описание = "Перевод выключен";
			Возврат Результат;
		КонецЕсли;
		Если ПустаяСтрока(Состояние.sel) Тогда
			Результат.Описание = "Сначала выберите карту для перевода";
			Возврат Результат;
		КонецЕсли;
		Если СыгратьКартуДурака(Состояние, Игрок, Состояние.sel, Истина) Тогда
			ЗаполнитьРезультатДурака(Результат, Состояние, Истина, "Перевод: " + Состояние.sel);
		Иначе
			Результат.Описание = Состояние.msg;
			Результат.Доска = СостояниеДуракаВСтроку(Состояние);
			Результат.Ок = Истина;
		КонецЕсли;
		Возврат Результат;
	КонецЕсли;

	Если Код = 91 Тогда
		Если Не МожноБитоДурака(Состояние, Игрок) Тогда
			Результат.Описание = "Сейчас нельзя сказать «Бито»/«Пас»";
			Возврат Результат;
		КонецЕсли;
		Если ВсеПарыПобитыДурака(Состояние) Тогда
			ЗавершитьРаундБитоДурака(Состояние);
			ЗаполнитьРезультатДурака(Результат, Состояние, Истина, "Бито");
		Иначе
			ЗавершитьВзятиеДурака(Состояние);
			ЗаполнитьРезультатДурака(Результат, Состояние, Истина, "Пас — карты взяты");
		КонецЕсли;
		Возврат Результат;
	КонецЕсли;

	Если Код = 92 Тогда
		Если Не МожноВзятьДурака(Состояние, Игрок) Тогда
			Результат.Описание = "Сейчас нельзя взять";
			Возврат Результат;
		КонецЕсли;
		Состояние.phase = "add";
		Состояние.turn = Состояние.att;
		Состояние.sel = "";
		Состояние.msg = "Защитник берёт — атакующий может подкинуть или пас";
		ЗаполнитьРезультатДурака(Результат, Состояние, Истина, "Взять");
		Возврат Результат;
	КонецЕсли;

	Результат.Описание = "Неизвестное действие";
	Возврат Результат;

Функция ХодБотаДурака(Доска, ЯИгрок0ДляБота) Экспорт

	Результат = НовыйРезультатХодаДурака(Доска);
	Состояние = СостояниеДуракаИзСтроки(Доска);
	Если Состояние.status <> "игра" Тогда
		Результат.Статус = Состояние.status;
		Результат.Описание = "Игра окончена";
		Возврат Результат;
	КонецЕсли;

	Бот = ?(ЯИгрок0ДляБота, 1, 0);
	Если Состояние.turn <> Бот Тогда
		Результат.Описание = "Сейчас не ход бота";
		Возврат Результат;
	КонецЕсли;

	Рука = РукаИгрокаДурака(Состояние, Бот);
	Козырь = Состояние.trump;

	Если Состояние.phase = "defend" И Бот = Состояние.def Тогда

		ИндексНепобитой = ИндексПервойНепобитойДурака(Состояние);
		Если ИндексНепобитой < 0 Тогда
			Результат.Описание = "Нечего отбивать";
			Возврат Результат;
		КонецЕсли;
		Атака = Состояние.table[ИндексНепобитой].a;
		Лучшая = "";
		ЛучшийРанг = 99;
		Для Каждого Код Из Рука Цикл
			Если КартаБьётДурака(Код, Атака, Козырь) Тогда
				Р = РангЧислоДурака(Код);
				Штраф = ?(МастьКартыДурака(Код) = Козырь, 20, 0);
				Оценка = Р + Штраф;
				Если Оценка < ЛучшийРанг Тогда
					ЛучшийРанг = Оценка;
					Лучшая = Код;
				КонецЕсли;
			КонецЕсли;
		КонецЦикла;

		Если Не ПустаяСтрока(Лучшая) И ЛучшийРанг <= 14 Тогда
			Если СыгратьКартуДурака(Состояние, Бот, Лучшая) Тогда
				ЗаполнитьРезультатДурака(Результат, Состояние, Истина, "Бот отбился: " + Лучшая);
				Возврат Результат;
			КонецЕсли;
		КонецЕсли;

		Если Состояние.perevod Тогда
			Для Каждого Код Из Рука Цикл
				Если МожноПеревестиДурака(Состояние, Бот, Код) Тогда
					Если СыгратьКартуДурака(Состояние, Бот, Код) Тогда
						ЗаполнитьРезультатДурака(Результат, Состояние, Истина, "Бот перевёл: " + Код);
						Возврат Результат;
					КонецЕсли;
				КонецЕсли;
			КонецЦикла;
		КонецЕсли;

		Если Не ПустаяСтрока(Лучшая) Тогда
			Если СыгратьКартуДурака(Состояние, Бот, Лучшая) Тогда
				ЗаполнитьРезультатДурака(Результат, Состояние, Истина, "Бот отбился: " + Лучшая);
				Возврат Результат;
			КонецЕсли;
		КонецЕсли;

		Состояние.phase = "add";
		Состояние.turn = Состояние.att;
		Состояние.sel = "";
		Состояние.msg = "Бот берёт карты";
		ЗаполнитьРезультатДурака(Результат, Состояние, Истина, "Бот берёт");
		Возврат Результат;

	КонецЕсли;

	Если (Состояние.phase = "attack" Или Состояние.phase = "add") И Бот = Состояние.att Тогда

		ВсеПобиты = ВсеПарыПобитыДурака(Состояние);
		РежимВзятия = (Состояние.phase = "add") И Не ВсеПобиты;

		Кандидаты = Новый Массив;
		Для Каждого Код Из Рука Цикл
			Если МожноСыгратьКартуДурака(Состояние, Бот, Код) Тогда
				Кандидаты.Добавить(Код);
			КонецЕсли;
		КонецЦикла;

		Если Кандидаты.Количество() > 0 Тогда
			Лучшая = Кандидаты[0];
			ЛучшийРанг = РангЧислоДурака(Лучшая) + ?(МастьКартыДурака(Лучшая) = Козырь, 20, 0);
			Для Индекс = 1 По Кандидаты.Количество() - 1 Цикл
				Код = Кандидаты[Индекс];
				Оценка = РангЧислоДурака(Код) + ?(МастьКартыДурака(Код) = Козырь, 20, 0);
				Если Оценка < ЛучшийРанг Тогда
					ЛучшийРанг = Оценка;
					Лучшая = Код;
				КонецЕсли;
			КонецЦикла;
			Если СыгратьКартуДурака(Состояние, Бот, Лучшая) Тогда
				ЗаполнитьРезультатДурака(Результат, Состояние, Истина, "Бот ходит: " + Лучшая);
				Возврат Результат;
			КонецЕсли;
		КонецЕсли;

		Если МожноБитоДурака(Состояние, Бот) Тогда
			Если ВсеПобиты Тогда
				ЗавершитьРаундБитоДурака(Состояние);
				ЗаполнитьРезультатДурака(Результат, Состояние, Истина, "Бот: бито");
			Иначе
				ЗавершитьВзятиеДурака(Состояние);
				ЗаполнитьРезультатДурака(Результат, Состояние, Истина, "Бот: пас (подкидывание)");
			КонецЕсли;
			Возврат Результат;
		КонецЕсли;

	КонецЕсли;

	Результат.Описание = "Бот не нашёл хода";
	Возврат Результат;

// Выводит игрока из матча (сдача / закрытие 1С). При 2+ оставшихся — партия продолжается.
Функция ВывестиИгрокаДурака(Доска, ИндексИгрока, Причина = "сдался") Экспорт

	Результат = Новый Структура("Доска,Статус,Описание,Ок,Продолжается", Доска, "игра", "", Ложь, Ложь);
	Состояние = СостояниеДуракаИзСтроки(Доска);
	Если Состояние.status <> "игра" Тогда
		Результат.Статус = Состояние.status;
		Результат.Описание = "Матч уже не идёт";
		Возврат Результат;
	КонецЕсли;
	Если Не ИгрокАктивенДурака(Состояние, ИндексИгрока) Тогда
		Результат.Статус = Состояние.status;
		Результат.Описание = "Игрок уже вне игры";
		Результат.Ок = Истина;
		Возврат Результат;
	КонецЕсли;

	Имя = ИмяИгрокаДурака(Состояние, ИндексИгрока);
	ТекстПричины = СокрЛП(Строка(Причина));
	Если ПустаяСтрока(ТекстПричины) Тогда
		ТекстПричины = "сдался";
	КонецЕсли;

	Если ТипЗнч(Состояние.out) <> Тип("Массив") Тогда
		Состояние.out = МассивБулевоДурака(Неопределено, ЧислоИгроковДурака(Состояние));
	КонецЕсли;
	Пока Состояние.out.Количество() <= ИндексИгрока Цикл
		Состояние.out.Добавить(Ложь);
	КонецЦикла;
	Состояние.out[ИндексИгрока] = Истина;
	УстановитьРукуИгрокаДурака(Состояние, ИндексИгрока, Новый Массив);
	Состояние.table = Новый Массив;
	Состояние.sel = "";

	Активные = Новый Массив;
	Н = ЧислоИгроковДурака(Состояние);
	Для Игрок = 0 По Н - 1 Цикл
		Если ИгрокАктивенДурака(Состояние, Игрок) Тогда
			Активные.Добавить(Игрок);
		КонецЕсли;
	КонецЦикла;

	Если Активные.Количество() <= 1 Тогда
		Если Активные.Количество() = 1 Тогда
			Победитель = Активные[0];
			Состояние.loser = ИндексИгрока;
			Состояние.msg = Имя + " " + ТекстПричины + ". Победа: " + ИмяИгрокаДурака(Состояние, Победитель);
			Если Победитель = 0 Тогда
				Состояние.status = "победаБелых";
			Иначе
				Состояние.status = "победаЧерных";
			КонецЕсли;
		Иначе
			Состояние.status = "ничья";
			Состояние.msg = Имя + " " + ТекстПричины + ". Ничья — игроков не осталось";
		КонецЕсли;
		Результат.Продолжается = Ложь;
	Иначе
		СтарыйAtt = Состояние.att;
		Если ИгрокАктивенДурака(Состояние, СтарыйAtt) Тогда
			Состояние.att = СтарыйAtt;
		Иначе
			Состояние.att = СледующийАктивныйДурака(Состояние, СтарыйAtt);
		КонецЕсли;
		Состояние.def = СледующийАктивныйДурака(Состояние, Состояние.att);
		Состояние.turn = Состояние.att;
		Состояние.phase = "attack";
		Состояние.msg = Имя + " " + ТекстПричины + ". Игра продолжается ("
			+ Формат(Активные.Количество(), "ЧГ=0") + " игроков). Атакует "
			+ ИмяИгрокаДурака(Состояние, Состояние.att);
		ПродлитьТаймерХодаДурака(Состояние);
		Результат.Продолжается = Истина;
	КонецЕсли;

	Результат.Доска = СостояниеДуракаВСтроку(Состояние);
	Результат.Статус = Состояние.status;
	Результат.Описание = Состояние.msg;
	Результат.Ок = Истина;
	Возврат Результат;

		КонецЕсли;
	КонецЦикла;
	Если Не ЕстьРанг Тогда
		Возврат Ложь;
	КонецЕсли;
	НовыйЗащ = СледующийАктивныйДурака(Состояние, Игрок);
	РукаНового = РукаИгрокаДурака(Состояние, НовыйЗащ);
	// После перевода на столе станет table+1 атак; у нового защитника должно хватить карт.
	Возврат РукаНового.Количество() >= Состояние.table.Количество() + 1;

КонецФункции

Функция МожноСыгратьКартуДурака(Состояние, Игрок, Код)

	Рука = РукаИгрокаДурака(Состояние, Игрок);
	Если Не ЕстьКартаВРукеДурака(Рука, Код) Тогда
		Возврат Ложь;
	КонецЕсли;

	Если Состояние.phase = "attack" И Игрок = Состояние.att Тогда
		Если Состояние.table.Количество() <> 0 Тогда
			Возврат Ложь;
		КонецЕсли;
		Возврат РукаИгрокаДурака(Состояние, Состояние.def).Количество() > 0;
	КонецЕсли;

	Если Состояние.phase = "defend" И Игрок = Состояние.def Тогда
		Если МожноПеревестиДурака(Состояние, Игрок, Код) Тогда
			Возврат Истина;
		КонецЕсли;
		Индекс = ИндексПервойНепобитойДурака(Состояние);
		Если Индекс < 0 Тогда
			Возврат Ложь;
		КонецЕсли;
		Возврат КартаБьётДурака(Код, Состояние.table[Индекс].a, Состояние.trump);
	КонецЕсли;

	Если Состояние.phase = "add" И Игрок = Состояние.att Тогда
		Если Не МожноДобавитьАтакуПоЛимитуДурака(Состояние) Тогда
			Возврат Ложь;
		КонецЕсли;
		Ранги = РангиНаСтолеДурака(Состояние);
		Возврат Ранги.Получить(РангСимволДурака(Код)) <> Неопределено;
	КонецЕсли;

	Возврат Ложь;

Функция СыгратьКартуДурака(Состояние, Игрок, Код, ПринудительныйПеревод = Ложь)

	Код = ВРег(Код);
	Если ПринудительныйПеревод Тогда
		Если Не МожноПеревестиДурака(Состояние, Игрок, Код)
			Или Не ЕстьКартаВРукеДурака(РукаИгрокаДурака(Состояние, Игрок), Код) Тогда
			Состояние.msg = "Этой картой перевести нельзя";
			Возврат Ложь;
		КонецЕсли;
	ИначеЕсли Не МожноСыгратьКартуДурака(Состояние, Игрок, Код) Тогда
		Состояние.msg = "Этой картой сейчас ходить нельзя";
		Возврат Ложь;
	КонецЕсли;

	Рука = РукаИгрокаДурака(Состояние, Игрок);

	Если ПринудительныйПеревод Тогда
		УдалитьКартуИзРукиДурака(Рука, Код);
		УстановитьРукуИгрокаДурака(Состояние, Игрок, Рука);

		Состояние.table.Добавить(Пара);
		Состояние.att = Игрок;
		Состояние.def = СледующийАктивныйДурака(Состояние, Игрок);

		УдалитьКартуИзРукиДурака(Рука, Код);
		УстановитьРукуИгрокаДурака(Состояние, Игрок, Рука);
		Пара = Новый Структура("a,d", Код, "");
		Состояние.table.Добавить(Пара);
		Состояние.phase = "defend";
		Состояние.turn = Состояние.def;
		Состояние.sel = "";
		Состояние.msg = "Защищайтесь";
		ПроверитьПобедуПослеХодаДурака(Состояние);
		Возврат Истина;
	КонецЕсли;

	Если Состояние.phase = "defend" Тогда
		ИндексНепобитой = ИндексПервойНепобитойДурака(Состояние);
		Бьёт = (ИндексНепобитой >= 0) И КартаБьётДурака(Код, Состояние.table[ИндексНепобитой].a, Состояние.trump);
		Перевод = МожноПеревестиДурака(Состояние, Игрок, Код);

		// «Сыграть» / слот «?» — сначала отбой; перевод только кнопкой «Перевод».
		Если Бьёт Тогда
			УдалитьКартуИзРукиДурака(Рука, Код);
			УстановитьРукуИгрокаДурака(Состояние, Игрок, Рука);
			Состояние.table[ИндексНепобитой].d = Код;
			Состояние.sel = "";
			Если ВсеПарыПобитыДурака(Состояние) Тогда
				Состояние.phase = "add";
				Состояние.turn = Состояние.att;
				Состояние.msg = "Можно подкинуть или бито";
			Иначе
				Состояние.turn = Состояние.def;
				Состояние.msg = "Отбейте следующую";
			КонецЕсли;
			ПроверитьПобедуПослеХодаДурака(Состояние);
			Возврат Истина;
		КонецЕсли;

		Если Перевод Тогда
			УдалитьКартуИзРукиДурака(Рука, Код);
			УстановитьРукуИгрокаДурака(Состояние, Игрок, Рука);
			Пара = Новый Структура("a,d", Код, "");
			Состояние.table.Добавить(Пара);
			Состояние.att = Игрок;
			Состояние.def = СледующийАктивныйДурака(Состояние, Игрок);
			Состояние.phase = "defend";
			Состояние.turn = Состояние.def;

			Состояние.msg = "Перевод на " + ИмяИгрокаДурака(Состояние, Состояние.def);
			ПроверитьПобедуПослеХодаДурака(Состояние);
			Возврат Истина;
		КонецЕсли;

		Состояние.msg = "Нельзя отбить/перевести этой картой";
		Возврат Ложь;
	КонецЕсли;

	Если Состояние.phase = "add" Тогда
		РежимВзятия = Не ВсеПарыПобитыДурака(Состояние);
		УдалитьКартуИзРукиДурака(Рука, Код);
		УстановитьРукуИгрокаДурака(Состояние, Игрок, Рука);
		Пара = Новый Структура("a,d", Код, "");
		Состояние.table.Добавить(Пара);
		Состояние.sel = "";
		Если РежимВзятия Тогда
			Состояние.phase = "add";
			Состояние.turn = Состояние.att;
			Состояние.msg = "Можно ещё подкинуть или пас";
		Иначе
			Состояние.phase = "defend";
			Состояние.turn = Состояние.def;
			Состояние.msg = "Защищайтесь";
		КонецЕсли;
		ПроверитьПобедуПослеХодаДурака(Состояние);
		Возврат Истина;
	КонецЕсли;

	Состояние.msg = "Неверная фаза";
	Возврат Ложь;

				Если Элемент.Свойство("u") Тогда
					Открыта = Элемент.u;

			Если Не ПустаяСтрока(Код) Тогда
				Результат.Добавить(КартаПасьянса(Код, Открыта));
			КонецЕсли;

		Возврат НегамаксШашки(Ход.ДоскаПосле, ХодБелых, Глубина, Альфа, Бета,
			Ход.ВСтрока, Ход.ВКолонка, НовыйЗапрет, Истина, ОбязательноеВзятие);
	КонецЕсли;

	Возврат -НегамаксШашки(Ход.ДоскаПосле, Не ХодБелых, Глубина - 1, -Бета, -Альфа,
		-1, -1, "|", Ложь, ОбязательноеВзятие);

КонецФункции

		Возврат ОценкаКонцаПартииДляСтороны(Статус, ХодБелых, Глубина);

		Возврат ОценкаПозицииШашки(Доска, ХодБелых);
	КонецЕсли;

	Ходы = ПолучитьХоды(Доска, ХодБелых, СтрокаКлетки, КолонкаКлетки, ЗапретВзятий, ТолькоВзятия, ОбязательноеВзятие);
	Если Ходы.Количество() = 0 Тогда
		Возврат ОценкаПозицииШашки(Доска, ХодБелых);
	КонецЕсли;

	Ходы = УпорядочитьХодыШашкиДляПоиска(Ходы);
	Лучший = -999999;
	Для Каждого Ход Из Ходы Цикл
		Балл = ОценкаХодаШашкиНегамакс(Ход, ХодБелых, ЗапретВзятий, ОбязательноеВзятие, Глубина, Альфа, Бета);
		Если Балл > Лучший Тогда
			Лучший = Балл;
		КонецЕсли;
		Если Лучший > Альфа Тогда
			Альфа = Лучший;
		КонецЕсли;
		Если Альфа >= Бета Тогда
			Прервать;
		КонецЕсли;
	КонецЦикла;
	Возврат Лучший;

Функция ОценкаПозицииШашки(Доска, СТочкиЗренияБелых) Экспорт

	Сумма = 0;
	Для Строка = 0 По 7 Цикл
		Для Колонка = 0 По 7 Цикл
			Символ = СимволКлетки(Доска, Строка, Колонка);
			Если Символ = "б" Тогда
				Сумма = Сумма + 100 + (7 - Строка) * 3;
			ИначеЕсли Символ = "Б" Тогда
				Сумма = Сумма + 280 + БонусЦентраКлетки(Строка, Колонка);
			ИначеЕсли Символ = "ч" Тогда
				Сумма = Сумма - 100 - Строка * 3;
			ИначеЕсли Символ = "Ч" Тогда
				Сумма = Сумма - 280 - БонусЦентраКлетки(Строка, Колонка);
			КонецЕсли;
		КонецЦикла;
	КонецЦикла;
	Если СТочкиЗренияБелых Тогда
		Возврат Сумма;
	КонецЕсли;
	Возврат -Сумма;

Функция ОценкаПозицииШахмат(Доска, СТочкиЗренияБелых) Экспорт

	Сумма = 0;
	Для Строка = 0 По 7 Цикл
		Для Колонка = 0 По 7 Цикл
			Символ = СимволКлетки(Доска, Строка, Колонка);
			Цена = ЦенаШахматнойФигуры(Символ);
			Если Цена = 0 Тогда
				Продолжить;
			КонецЕсли;
			Позиция = БонусПозицииШахматнойФигуры(Символ, Строка, Колонка);
			Если ЦветШахматнойФигуры(Символ) = "б" Тогда
				Сумма = Сумма + Цена * 100 + Позиция;
			Иначе
				Сумма = Сумма - Цена * 100 - Позиция;
			КонецЕсли;
		КонецЦикла;
	КонецЦикла;
	Если КорольПодШахом(Доска, Истина) Тогда
		Сумма = Сумма - 35;
	КонецЕсли;
	Если КорольПодШахом(Доска, Ложь) Тогда
		Сумма = Сумма + 35;
	КонецЕсли;
	Если СТочкиЗренияБелых Тогда
		Возврат Сумма;
	КонецЕсли;
	Возврат -Сумма;

Функция ЦенаШахматнойФигуры(Символ)

	Вид = НРег(Символ);
	Если Вид = "p" Тогда
		Возврат 1;
	ИначеЕсли Вид = "n" Или Вид = "b" Тогда
		Возврат 3;
	ИначеЕсли Вид = "r" Тогда
		Возврат 5;
	ИначеЕсли Вид = "q" Тогда
		Возврат 9;
	ИначеЕсли Вид = "k" Тогда
		Возврат 100;
	КонецЕсли;
	Возврат 0;

Функция БонусПозицииШахматнойФигуры(Символ, Строка, Колонка)

	Вид = НРег(Символ);
	Центр = БонусЦентраКлетки(Строка, Колонка);
	Если Вид = "p" Тогда
		Если ЦветШахматнойФигуры(Символ) = "б" Тогда
			Возврат (7 - Строка) * 4 + Центр;
		КонецЕсли;
		Возврат Строка * 4 + Центр;
	ИначеЕсли Вид = "n" Или Вид = "b" Тогда
		Возврат Центр * 2;
	ИначеЕсли Вид = "k" Тогда
		// В дебюте/миттельшпиле королю спокойнее у края.
		Возврат -Центр;
	КонецЕсли;
	Возврат Центр;

Функция БонусЦентраКлетки(Строка, Колонка)

	ДС = Мин(Строка, 7 - Строка);
	ДК = Мин(Колонка, 7 - Колонка);
	Возврат ДС + ДК;
